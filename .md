public static class CreateBranch
{
    public sealed record Request
    {
        public string BranchCode { get; init; } = string.Empty;
        public string BranchName { get; init; } = string.Empty;
        public string? AddressLine1 { get; init; }
        public string? AddressLine2 { get; init; }
        public string? City { get; init; }
        public string? Area { get; init; }
        public string? Phone { get; init; }
        public string? Email { get; init; }
        public string? ManagerName { get; init; }
        public bool IsActive { get; init; }
        public string? ImageUrl { get; init; }
        public string? ImageBase64 { get; init; }
        public string? ImageContentType { get; init; }
    }

    public sealed record Response
    {
        public Guid Id { get; init; }
        public string BranchCode { get; init; } = string.Empty;
        public string BranchName { get; init; } = string.Empty;
        public string? AddressLine1 { get; init; }
        public string? AddressLine2 { get; init; }
        public string? City { get; init; }
        public string? Area { get; init; }
        public string? Phone { get; init; }
        public string? Email { get; init; }
        public string? ManagerName { get; init; }
        public bool IsActive { get; init; }
        public string? ImageUrl { get; init; }
    }

    public sealed record Command(
        string BranchCode, string BranchName, string? AddressLine1, string? AddressLine2, string? City, 
        string? Area, string? Phone, string? Email, string? ManagerName, bool IsActive,
        string? ImageUrl, string? ImageBase64, string? ImageContentType);

    public sealed class Validator : AbstractValidator<Request>
    {
        public Validator() {
            RuleFor(x => x.BranchCode).NotEmpty();
            RuleFor(x => x.BranchName).NotEmpty();
        }
    }

    public sealed class Handler
    {
        private readonly SuperShopDbContext _context;

        public Handler(SuperShopDbContext context)
        {
            _context = context;
        }

        public async Task<Result<Response>> Handle(Command command, CancellationToken ct)
        {
            var exists = await _context.Branches
                .AnyAsync(x => x.BranchCode == command.BranchCode && !x.IsDeleted, ct);

            if (exists)
            {
                return Result<Response>.Failure(Error.Conflict($"Branch with BranchCode '{command.BranchCode}' already exists."));
            }

            var item = new Branch
            {
                BranchCode = command.BranchCode,
                BranchName = command.BranchName,
                AddressLine1 = command.AddressLine1,
                AddressLine2 = command.AddressLine2,
                City = command.City,
                Area = command.Area,
                Phone = command.Phone,
                Email = command.Email,
                ManagerName = command.ManagerName,
                IsActive = command.IsActive,
                ImageUrl = command.ImageUrl
            };

            if (!string.IsNullOrEmpty(command.ImageBase64))
            {
                try
                {
                    item.ImageData = Convert.FromBase64String(command.ImageBase64);
                    item.ImageContentType = command.ImageContentType;
                    item.ImageUrl = null;
                }
                catch { /* Ignore invalid base64 */ }
            }

            _context.Branches.Add(item);
            await _context.SaveChangesAsync(ct);

            return Result<Response>.Success(item.Adapt<Response>());
        }
    }
}

            }

            return Result<Response>.Success(item);
        }
    }

public static class GetBranches
{
    public sealed record Request(int PageIndex = 0, int PageSize = 10, string? Search = null);

    public sealed record Response
    {
        public Guid Id { get; init; }
        public string BranchCode { get; init; } = string.Empty;
        public string BranchName { get; init; } = string.Empty;
        public string? AddressLine1 { get; init; }
        public string? AddressLine2 { get; init; }
        public string? City { get; init; }
        public string? Area { get; init; }
        public string? Phone { get; init; }
        public string? Email { get; init; }
        public string? ManagerName { get; init; }
        public bool IsActive { get; init; }
    }

    public sealed record Query(int PageIndex, int PageSize, string? Search);

    public sealed class Handler
    {
        private readonly SuperShopDbContext _context;

        public Handler(SuperShopDbContext context)
        {
            _context = context;
        }

        public async Task<Result<PagedResult<Response>>> Handle(Query query, CancellationToken ct)
        {
            var dbQuery = _context.Branches
                .Where(x => !x.IsDeleted)
                .AsNoTracking();

            if (!string.IsNullOrWhiteSpace(query.Search))
            {
                dbQuery = dbQuery.Where(x => x.BranchName.Contains(query.Search) || x.BranchCode.Contains(query.Search));
            }

            var totalCount = await dbQuery.CountAsync(ct);
            var items = await dbQuery
                .Skip(query.PageIndex * query.PageSize)
                .Take(query.PageSize)
                .ProjectToType<Response>()
                .ToListAsync(ct);

            return Result<PagedResult<Response>>.Success(new PagedResult<Response>(items, totalCount, query.PageIndex, query.PageSize));
        }
    }
}

    {
        var group = app.MapGroup("/api/customers").WithTags("Customers");

        group.MapGet("/", async (
            [AsParameters] GetCustomers.Request request, 
            [FromServices] GetCustomers.Handler handler, 
            CancellationToken ct) =>
        {
            var query = new GetCustomers.Query(request.PageIndex, request.PageSize, request.Search);
            var result = await handler.Handle(query, ct);
            return result.ToHttpResult();
        })
        .WithName("GetCustomers")
        .WithSummary("Get paginated customers");

        group.MapGet("/{id:guid}", async (
            Guid id, 
            [FromServices] GetCustomerById.Handler handler, 
            CancellationToken ct) =>
        {
            var result = await handler.Handle(new GetCustomerById.Query(id), ct);
            return result.ToHttpResult();
        })
        .WithName("GetCustomerById")
        .WithSummary("Get customers by id");

        group.MapPost("/", async (
            [FromBody] CreateCustomer.Request request,
